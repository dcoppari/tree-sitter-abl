================================================================================
SQL CURSOR - full pattern: DECLARE OPEN REPEAT FETCH CLOSE
================================================================================

declare myRS cursor for
    select max(regnro) + 1
      from FACENCAB
     where FACENCAB.siscod = wsiscod.

open myRS.
repeat:
    fetch myRS into lastNro.
end.
close myRS.
